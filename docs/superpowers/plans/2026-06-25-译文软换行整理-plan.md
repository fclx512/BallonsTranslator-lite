```python
import re

QUOTE_PAIRS = [("「", "」")]

def normalize_softbreaks(text: str) -> str:
    """按「」分块：块内 \\n → 空格，块间单 \\n 连接。
       不含「」的文本整体 \\n → 空格。
       幂等：再跑一次结果不变。"""
    if not isinstance(text, str) or text == "":
        return text

    # 无引号段整体 \n → 空格（走通用路径）
    has_any_quote = any(open_q in text for open_q, _ in QUOTE_PAIRS)
    if not has_any_quote:
        return _collapse_newlines(text)

    # 有引号：按最外层「」分块
    segments = _split_by_outer_quotes(text)
    result_parts = []
    for seg_type, content in segments:
        if seg_type == "quote":
            result_parts.append(f"「{_collapse_newlines(content)}」")
        else:
            result_parts.append(_collapse_newlines(content))

    # 合并：引号段之间补 \n，但避免与普通段合并
    return "".join(_insert_joins(result_parts))


def _collapse_newlines(s: str) -> str:
    """将连续 \\n/空格压缩为单个空格，剥首尾空白。"""
    import re
    s = re.sub(r"[ \t]*\n+[ \t]*", " ", s)
    s = re.sub(r" {2,}", " ", s)
    return s.strip()


def _split_by_outer_quotes(text: str) -> list:
    """按最外层「」切分。返回 [(type, content), ...]，type='quote'|'plain'。
       嵌套「」取最外层，内层保留字面。"""
    import re
    segments = []
    pos = 0
    # 匹配「...」嵌套用递进法：找到「后，逐字符计数跟踪嵌套深度，遇到匹配的」结束
    while pos < len(text):
        open_q = "「"
        close_q = "」"
        if text[pos] == open_q:
            depth = 1
            start = pos + 1
            pos += 1
            while pos < len(text) and depth > 0:
                if text[pos] == open_q:
                    depth += 1
                elif text[pos] == close_q:
                    depth -= 1
                pos += 1
            # pos 此时指向「」后的字符
            inner = text[start:pos-1]  # 去掉外层「」
            segments.append(("quote", inner))
        else:
            start = pos
            while pos < len(text) and text[pos] != open_q:
                pos += 1
            segments.append(("plain", text[start:pos]))
    return segments


def _insert_joins(parts: list) -> list:
    """在靠近的引号段和普通段之间填内容，既不产生多余空格也不丢掉必要的连接。
       主要是保证「」段后紧跟的纯文本不凭空加 \\n。逻辑：
       - 如果段已经是空串，跳过
       - 引号段后如果紧接非空普通段（或是结尾），追加到 result
       - 完全不额外加 \\n——_split_by_outer_quotes 返回的已经是原位置内容"""
    # 实际上_split_by_outer_quotes保留了位置关系，拼接时不需要额外加join
    # 只需要保证「片尾」紧跟的纯文本不带多余空格
    return parts  # 默认按原样拼接


# 对 _insert_joins 修正：实际需要处理的是「」段间用 \n 连接
# 重写：

def _insert_joins(parts: list) -> list:
    """核心逻辑：
       - 相邻两个引号段之间用 \\n 连接（它们原本被 \n 分隔）
       - 引号段后紧接普通段、或反之，直接拼接（普通段清空了\n变成空格）"""
    result = []
    for i, (typ, content) in enumerate(parts):
        if typ == "quote":
            result.append(f"「{content}」")
        else:
            result.append(content)
        # 判断下一个段前是否需要加 \n
        if i + 1 < len(parts):
            next_typ = parts[i + 1][0]
            if typ == "quote" and next_typ == "quote":
                result.append("\n")
            # 其他情况：普通段已清掉\n，直接拼接无问题
    return result
```

```python
# ---------- NormalizeBreaksCommand ----------

class NormalizeBreaksCommand(QUndoCommand):
    """整理换行的 undo 命令。一次 push，Ctrl+Z 全部回退。"""

    def __init__(
        self,
        proj: "ProjImgTrans",
        scene_manager,
        changes: List[dict],
        parent=None,
    ):
        super().__init__(parent)
        self._first_redo = True
        self.proj = proj
        self.sm = scene_manager
        self.changes = changes  # 每项: {pagename, block_idx, old_translation, old_rich_text, new_text, squeeze}

    def _apply(self, mode: str):
        """mode='new': 应用新文本；mode='old': 还原旧文本。"""
        for chg in self.changes:
            pname = chg["pagename"]
            bidx = chg["block_idx"]
            if mode == "new":
                new_text = chg["new_text"]
                squeeze = chg.get("squeeze", False)
            else:
                new_text = chg.get("old_translation", "")
                squeeze = False  # undo 不 squeeze，恢复旧框

            blk = self.proj.pages[pname][bidx]
            current_pname = self.proj.idx2pagename(self.proj.current_page_idx)
            if pname != current_pname:
                # 非当前页：只写 blk.translation
                blk.translation = new_text
                blk.rich_text = ""
                # 还原时恢复旧 rich_text
                if mode == "old":
                    blk.rich_text = chg["old_rich_text"]
            else:
                # 当前页：通过 TextBlkItem 写
                item = self.sm._find_blk_item(bidx)
                if item is None:
                    continue
                # 存旧矩形
                if mode == "new":
                    # 第一次 redo 时，item 的当前状态已在 init 时存好
                    pass
                elif mode == "old":
                    old_rect = chg.get("old_rect", None)
                    if old_rect:
                        item.setRect(old_rect, repaint=False)

                # 写文本
                item.setPlainTextAndKeepUndoStack(new_text)
                item.set_fontformat(item.fontformat, set_char_format=True)

                # squeeze
                if mode == "new" and squeeze:
                    item.squeezeBoundingRect(repaint=True)

                item.repaint_background()

    def redo(self):
        if self._first_redo:
            self._first_redo = False
            return
        self._apply("new")

    def undo(self):
        self._apply("old")
```

> **说明：** `_find_blk_item` 是 `SceneTextManager` 上一个已知的辅助函数（`fontstyle_manager_commands.py:126-134` 有相同函数签名）。如果 `SceneTextManager` 不暴露它，需要在 `SceneTextManager` 上加一个公用方法或在 `NormalizeBreaksCommand.__init__` 中用导引生成索引。

```python
from PyQt6.QtWidgets import (
    QDialog, QVBoxLayout, QHBoxLayout, QCheckBox,
    QPushButton, QScrollArea, QWidget, QGroupBox,
)
from PyQt6.QtCore import Qt
from utils.text_normalize import normalize_softbreaks

class NormalizeBreaksDialog(QDialog):
    """批量整理换行对话框。"""

    def __init__(self, proj, scene_manager, parent=None):
        super().__init__(parent)
        self.proj = proj
        self.sm = scene_manager
        self.processed_count = 0
        self.skipped_count = 0
        self._changes = []
        self._setup_ui()
        self._populate_pages()

    def _setup_ui(self):
        self.setWindowTitle(self.tr("批量整理换行"))
        self.setMinimumWidth(400)
        layout = QVBoxLayout(self)

        # 全部页复选框
        self.all_check = QCheckBox(self.tr("全部页"), self)
        self.all_check.setChecked(True)
        self.all_check.toggled.connect(self._on_all_toggled)
        layout.addWidget(self.all_check)

        # 页码列表（滚动区域）
        self.scroll = QScrollArea(self)
        self.scroll.setWidgetResizable(True)
        self.page_container = QWidget()
        self.page_layout = QVBoxLayout(self.page_container)
        self.page_layout.setContentsMargins(0, 0, 0, 0)
        self.page_layout.setSpacing(2)
        self.scroll.setWidget(self.page_container)
        layout.addWidget(self.scroll, stretch=1)

        # 勾选项
        self.squeeze_check = QCheckBox(self.tr("完成后自动收缩框"), self)
        self.squeeze_check.setChecked(False)
        layout.addWidget(self.squeeze_check)

        # 按钮
        btn_layout = QHBoxLayout()
        apply_btn = QPushButton(self.tr("应用"), self)
        cancel_btn = QPushButton(self.tr("取消"), self)
        apply_btn.clicked.connect(self._on_apply)
        cancel_btn.clicked.connect(self.reject)
        btn_layout.addStretch()
        btn_layout.addWidget(apply_btn)
        btn_layout.addWidget(cancel_btn)
        layout.addLayout(btn_layout)

    def _populate_pages(self):
        self.page_checks = []
        for i in range(self.proj.num_pages):
            pname = self.proj.idx2pagename(i)
            cb = QCheckBox(f"{self.tr('第')} {i} {self.tr('页')} — {pname}", self)
            cb.setEnabled(False)  # 默认全部页勾选→禁用
            self.page_layout.addWidget(cb)
            self.page_checks.append(cb)
        self.page_layout.addStretch()

    def _on_all_toggled(self, checked: bool):
        for cb in self.page_checks:
            cb.setEnabled(not checked)
            if checked:
                cb.setChecked(False)  # 视觉上全部不勾（实际按"所有页"处理）

    def _on_apply(self):
        self._changes = []
        self.processed_count = 0
        self.skipped_count = 0

        if self.all_check.isChecked():
            page_indices = range(self.proj.num_pages)
        else:
            page_indices = [i for i, cb in enumerate(self.page_checks) if cb.isChecked()]

        squeeze = self.squeeze_check.isChecked()

        for pi in page_indices:
            pname = self.proj.idx2pagename(pi)
            blocks = self.proj.pages.get(pname, [])
            for bidx, blk in enumerate(blocks):
                if blk.vertical:
                    self.skipped_count += 1
                    continue
                if not isinstance(blk.translation, str) or blk.translation == "":
                    continue
                new_text = normalize_softbreaks(blk.translation)
                if new_text == blk.translation:
                    continue

                # 对当前页块，记录旧 HTML / 矩形
                is_current = (pname == self.proj.idx2pagename(self.proj.current_page_idx))
                old_rich = blk.rich_text
                old_rect = list(blk._bounding_rect) if blk._bounding_rect else None
                if is_current and self.sm is not None:
                    item = self.sm.blk_item_map.get(bidx)
                    if item is not None:
                        old_rich = item.toHtml()
                        old_rect = list(item.absBoundingRect(qrect=True).getRect()) if item.absBoundingRect(qrect=True) else None

                self._changes.append({
                    "pagename": pname,
                    "block_idx": bidx,
                    "old_translation": blk.translation,
                    "old_rich_text": old_rich,
                    "old_rect": old_rect,
                    "new_text": new_text,
                    "squeeze": squeeze,
                })
                self.processed_count += 1

        self.accept()

    def get_changes(self) -> list:
        return self._changes
```
